Нужно запустить сервер 
npm instal

открыть второй терминал в эту же директорию
# 1. Сборка проекта
npm run build

# 2. Запуск сервера
npm start

# 3. Проверка корневого эндпоинта
curl http://localhost:4000/

# 4. Клиенты (/client)
# Получение списка клиентов
curl http://localhost:4000/client

# Создание нового клиента
curl -X POST http://localhost:4000/client -H "Content-Type: application/json" -d "{\"name\":\"John Doe\",\"email\":\"john@example.com\",\"phone\":\"123456789\"}"

# Получение клиента по ID (замените <client-id> на ID созданного клиента)
curl http://localhost:4000/client/<client-id>

# Обновление клиента
curl -X PUT http://localhost:4000/client/<client-id> -H "Content-Type: application/json" -d "{\"name\":\"John Smith\"}"

# Получение заказов клиента
curl http://localhost:4000/client/<client-id>/order

# Удаление клиента
curl -X DELETE http://localhost:4000/client/<client-id>

# 5. Заказы (/order)
# Создание клиента для теста
curl -X POST http://localhost:4000/client -H "Content-Type: application/json" -d "{\"name\":\"Jane Doe\",\"email\":\"jane@example.com\",\"phone\":\"987654321\"}"

# Создание нового заказа (замените <client-id> на ID клиента)
curl -X POST http://localhost:4000/order -H "Content-Type: application/json" -d "{\"clientId\":\"<client-id>\",\"products\":[],\"status\":\"pending\",\"total\":0}"
